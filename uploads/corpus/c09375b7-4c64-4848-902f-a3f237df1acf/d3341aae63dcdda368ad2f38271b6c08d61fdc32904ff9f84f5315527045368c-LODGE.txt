	•	Lens gates (bias, opt-in, evidence grade)

LOGE does not need “truth.” It governs whether actions are permitted under evidence quality and policy.

3.1 sha256 helper

export async function sha256Hex(data: string): Promise<string> {
  const enc = new TextEncoder();
  const buf = await crypto.subtle.digest("SHA-256", enc.encode(data));
  return [...new Uint8Array(buf)].map(b => b.toString(16).padStart(2, "0")).join("");
}

3.2 Evidence pack canonicalization

packages/core/src/hashing/canonicalize.ts

import { EvidencePack, ExtractedItem } from "../types/evidence";

export function canonicalizeEvidencePack(pack: EvidencePack): string {
  const prepItem = (i: ExtractedItem) => ({
    itemId: i.itemId,
    type: i.type,
    v: i.normalizedValue ?? i.value,
    status: i.status,
    confScore: i.confidence.score,
    confBand: i.confidence.band,
    anchors: i.anchors
      .map(a => `${a.docId}:${a.segId}:${String(a.charStart).padStart(10,"0")}:${String(a.charEnd).padStart(10,"0")}`)
      .sort(),
    flags: i.flags,
  });

  const canonical = {
    packId: pack.packId,
    parentRevisionId: pack.parentRevisionId,
    docs: pack.docs.map(d => ({
      docId: d.docId,
      segments: d.segments.map(s => ({
        segId: s.segId,
        charStart: s.charStart,
        charEnd: s.charEnd,
        // IMPORTANT: segment text IS part of canonical evidence (anchor firewall depends on it)
        text: s.text,
      })).sort((a,b)=>a.segId.localeCompare(b.segId)),
    })).sort((a,b)=>a.docId.localeCompare(b.docId)),
    items: pack.items.map(prepItem).sort((a,b)=>a.itemId.localeCompare(b.itemId)),
    flags: pack.flags,
    // Exclude: revisionId, lensLog timestamps, runIds (audit metadata, not evidence state)
  };

  return JSON.stringify(canonical);
}

import { ActionProposal } from "../types/proposal";

export function normalizeProposal(proposal: ActionProposal): unknown {
  // Remove volatile fields; keep stable content only
  const base = {
    actionType: proposal.actionType,
    requestedBy: proposal.requestedBy,
    // requestedAt excluded from hash to prevent variance
    correlationId: proposal.correlationId ?? null,
    payload: proposal.payload,
  };

  // Canonical sort for arrays where order should not matter
  if (proposal.actionType === "PROMOTE_CLAIM") {
    return {
      ...base,
      payload: {
        ...proposal.payload,
        claimItemIds: [...proposal.payload.claimItemIds].sort(),
      },
    };
  }

  return base;
}

inv.identity_stability (ID → signature must not drift)

packages/core/src/invariants/inv_identity_stability.ts

import { InvariantContext, InvariantResult } from "../types/invariants";

function itemSignature(i: any): string {
  const anchors = i.anchors.map((a: any)=>`${a.docId}:${a.segId}:${a.charStart}-${a.charEnd}`).sort().join(",");
  const v = i.normalizedValue ?? i.value;
  return `${i.type}|${v}|${i.status}|${anchors}`;
}

export function inv_identity_stability(ctx: InvariantContext): InvariantResult {
  const map = new Map<string, string>();
  for (const item of ctx.pack.items) {
    const sig = itemSignature(item);
    const prior = map.get(item.itemId);
    if (prior && prior !== sig) {
      return {
        id: "inv.identity_stability",
        ok: false,
        severity: "CRITICAL",
        detail: `Identity drift: itemId ${item.itemId} maps to multiple signatures.`,
      };
    }
    map.set(item.itemId, sig);
  }
  return { id: "inv.identity_stability", ok: true, severity: "CRITICAL" };
}

inv.anchor_firewall (offset-value validation)

packages/core/src/invariants/inv_anchor_firewall.ts

import { InvariantContext, InvariantResult } from "../types/invariants";

export function inv_anchor_firewall(ctx: InvariantContext): InvariantResult {
  for (const item of ctx.pack.items) {
    for (const a of item.anchors) {
      const doc = ctx.pack.docs.find(d => d.docId === a.docId);
      if (!doc) return { id:"inv.anchor_firewall", ok:false, severity:"CRITICAL", detail:`Missing docId ${a.docId}` };

      const seg = doc.segments.find(s => s.segId === a.segId);
      if (!seg) return { id:"inv.anchor_firewall", ok:false, severity:"CRITICAL", detail:`Missing segId ${a.segId}` };

      const localStart = a.charStart - seg.charStart;
      const localEnd = a.charEnd - seg.charStart;

      if (localStart < 0 || localEnd > seg.text.length || localStart >= localEnd) {
        return { id:"inv.anchor_firewall", ok:false, severity:"CRITICAL", detail:`Invalid local offsets for itemId ${item.itemId}` };
      }

      const actual = seg.text.slice(localStart, localEnd);

      // Quotes must match exactly; others must contain value (conservative)
      const value = item.normalizedValue ?? item.value;
      const ok =
        item.type === "quote" ? actual === item.value :
        actual.includes(item.value) || actual.includes(value);

      if (!ok) {
        return { id:"inv.anchor_firewall", ok:false, severity:"CRITICAL", detail:`Anchor mismatch for itemId ${item.itemId}` };
      }
    }
  }
  return { id:"inv.anchor_firewall", ok:true, severity:"CRITICAL" };
}

export function inv_opt_in(ctx: InvariantContext): InvariantResult {

import { EvidencePack } from "../types/evidence";
import { ActionProposal } from "../types/proposal";
import { Policy } from "../types/policy";
import { sha256Hex } from "../hashing/sha256";
import { canonicalizeEvidencePack } from "../hashing/canonicalize";
import { normalizeProposal } from "./normalizePayload";

import { inv_identity_stability } from "../invariants/inv_identity_stability";
import { inv_anchor_firewall } from "../invariants/inv_anchor_firewall";
import { inv_claim_gate_bias } from "../invariants/inv_claim_gate_bias";
import { inv_opt_in } from "../invariants/inv_opt_in";

export async function evaluateProposal(args: {
  proposal: ActionProposal;
  pack: EvidencePack;
  policy: Policy;
}){
  const normalized = normalizeProposal(args.proposal);
  const payloadHash = await sha256Hex(JSON.stringify(normalized));

  const packHash = await sha256Hex(canonicalizeEvidencePack(args.pack));
  const policyHash = await sha256Hex(JSON.stringify(args.policy)); // canonicalize similarly if needed

  const ctx = { proposal: args.proposal, pack: args.pack, policy: args.policy, packHash, payloadHash, policyHash };

  const invariantResults = [
    inv_identity_stability(ctx as any),
    inv_anchor_firewall(ctx as any),
    inv_claim_gate_bias(ctx as any),
    inv_opt_in(ctx as any),
  ];

  const ok = invariantResults.every(r => r.ok);

  return { ok, invariantResults, packHash, payloadHash, policyHash, normalizedProposal: normalized };
}

6) PermitToken Issuance (Cryptographic)

Use ED25519 (preferred) or HMAC-SHA256 for v1.
Below shows HMAC for minimal dependency; swap later.

packages/core/src/governor/issuePermit.ts

import { PermitToken } from "../types/permit";
import { sha256Hex } from "../hashing/sha256";

// For HMAC v1, signature = sha256(secret + bodyHash). Replace with ED25519 later.
export async function signToken(body: Omit<PermitToken, "signature">, secret: string): Promise<PermitToken> {
  const bodyStr = JSON.stringify(body);
  const bodyHash = await sha256Hex(bodyStr);
  const signature = await sha256Hex(`${secret}|${bodyHash}`);
  return { ...body, signature };
}

export async function verifyToken(token: PermitToken, secret: string): Promise<boolean> {
  const { signature, ...body } = token;
  const candidate = await signToken(body as any, secret);
  return candidate.signature === signature;
}

7.3 /execute logic

packages/server/src/routes/execute.ts (pseudocode-ish but implementable)

import { verifyToken } from "@loge/core/governor/issuePermit";
import { sha256Hex } from "@loge/core/hashing/sha256";
import { canonicalizeEvidencePack } from "@loge/core/hashing/canonicalize";

export async function executeRoute(req, res, deps){
  const { token, proposal, pack } = req.body;

  // 1) Verify signature
  const okSig = await verifyToken(token, deps.signingSecret);
  if (!okSig) return res.status(401).send({ error: "Invalid signature" });

  // 2) Verify decision + expiry
  if (token.decision !== "APPROVE") return res.status(403).send({ error: "Token is not APPROVE" });
  if (Date.now() > Date.parse(token.expiresAt)) return res.status(403).send({ error: "Token expired" });

  // 3) Verify packHash matches current pack state
  const packHashNow = await sha256Hex(canonicalizeEvidencePack(pack));
  if (packHashNow !== token.packHash) return res.status(409).send({ error: "Evidence drift: packHash mismatch" });

  // 4) Verify payloadHash matches normalized proposal
  const payloadHashNow = await sha256Hex(JSON.stringify(deps.normalizeProposal(proposal)));
  if (payloadHashNow !== token.payloadHash) return res.status(409).send({ error: "Payload drift: payloadHash mismatch" });

  // 5) Replay protection
  const prior = await deps.ledger.getExecution(token.tokenId);
  if (prior) return res.status(409).send({ error: "Replay detected: token already executed" });

  // 6) Execute tool by actionType
  const result = await deps.tools.executeByActionType(proposal.actionType, proposal.payload, { pack });

  // 7) Persist execution
  await deps.ledger.recordExecution(token.tokenId, proposal.actionType, result);

  return res.send({ ok: true, result });
}

## 4) Non-Negotiable Invariants (The “Guardrails of Reality”)
These are hard blocks. If violated, the system must **refuse**.

### CRITICAL
1. **inv.identity_stability**  
   Same `itemId` must never map to a different `itemSignature`. If it does → drift/collision → HALT.

2. **inv.anchor_firewall**  
   Anchors must resolve to the expected text in the source segment:
   - Quotes: exact match (`actual === value`)
   - Non-quotes: `actual.includes(value)` or `actual.includes(normalizedValue)`
   If mismatch → DISCARD item or REFUSE action depending on layer.

3. **inv.claim_gate_bias**  
   If `pack.flags.bias_flagged_critical === true`, then **constructive outputs** (claims) must be blocked and **PROMOTE_CLAIM must refuse**.

4. **inv.opt_in_discipline**  
   Constructive actions require explicit **human policy opt-in** for that packId.

### HIGH
5. **inv.hash_chain_integrity**  
   LensLog must form an unbroken chain: each revision’s outputHash must match computed hash for that pack content.

6. **inv.replay_protection**  
   A tokenId may execute at most once. Second attempt → 409.

## 5) Evidence Data Model (ExtractionPack — Immutable Revision Chain)
### 5.1 ExtractionPack (Core)
- `packId`: stable logical identifier
- `revisionId`: immutable revision head
- `parentRevisionId`: previous revision pointer
- `docs[]`: SourceDocs containing rawText + segments
- `items[]`: extracted items with anchors
- `flags`: pack-level diagnostics (e.g., bias_flagged_critical)

### 5.2 Anchor Contract (Mandatory)
Every item must have at least one anchor:
- `docId`, `segId`
- global `charStart`, global `charEnd`
- snippet (cache only; source-of-truth is segment text)

### 5.3 Deterministic Segment IDs
Segments use deterministic spans:
- `segId = ${docId}:${start}-${end}`
- Stability guarantee if doc unchanged and segmenter deterministic

## 6) Hashing + Canonicalization (Collision-Proof)
### 6.1 Pack Hash (packHash)
Must ignore volatile fields:
- timestamps
- runIds
- lensLog
Only include:
- packId, parentRevisionId
- docs (docId + segment ids in sorted order, optionally segment spans/text)
- items canonicalized (type, normalized/value, anchors sorted, confidence band/score if treated as content)
- pack flags

### 6.2 Item Identity (itemId)
**Collision-proof rule:** itemId MUST incorporate span boundaries.
Canonical formula:

sha256(
type
| normalizedValueOrValue
| docId
| segId
| charStart
| charEnd
)

(If you omit charEnd, span collisions are possible → corruption.)

### 6.3 Item Signature (anti-drift)
`itemSignature` is the semantic commitment an ID represents:

type | normalizedValueOrValue | status | sortedAnchors(docId:segId:start-end)

If same itemId maps to different signature → identity drift CRITICAL.

## 7) The Six Lenses (Evidence Plane)
All lenses must:
- return a **new** pack revision
- set `revisionId = runId()`
- set `parentRevisionId = prior revisionId`
- append lensLog entry with inputHashes/outputHash
- on refusal, still produce a revision and lensLog event

### Lens 1 — `core.segment_anchor.v1`
Hybrid paragraph/sentence deterministic spanning segmenter with overlap guard:
- `OVERLAP_CHARS = 60` to avoid boundary splits
- stable segId spans
- refusal if text too short or no segments

### Lens 2 — `core.typed_extract.v1`
LLM extraction with strict schema firewall:
- only outputs: entity/quote/metric (primitives only)
- MUST include segment-local offsets
- translate to global offsets: `globalStart = seg.charStart + localStart`
- **Offset-Value Firewall**:
  - quotes require exact match
  - non-quotes require substring containment
- discard-first on schema mismatch

### Lens 3 — `core.canonicalize_dedupe.v1`
Deterministic canonicalization + clustering:
- compute `normalizedValue`
- compute `clusterId = sha256(type|normalizedValue)`
- representative item: lexicographically lowest itemId
- annotate items; never merge/drop provenance

### Lens 4 — `core.evidence_grade.v1`
Deterministic “truthiness” grading:
- uses structural heuristics (consensus, multi-seg, flags)
- snap numeric scores to 2 decimals
- outputs transparent rationale strings

### Lens 5 — `diagnostic.bias_coverage.v1`
Flags fragility:
- single-source dominance
- low context (pronoun-heavy, no entity proxy)
- bias critical threshold:
  - top segments cover >=80% anchors while <=10% of total segments

### Lens 6 — `core.precision_claims.v1`
Constructive, opt-in only:
- refuses if bias critical
- uses only HIGH primitives
- claim must reference supporting itemIds + cluster provenance
