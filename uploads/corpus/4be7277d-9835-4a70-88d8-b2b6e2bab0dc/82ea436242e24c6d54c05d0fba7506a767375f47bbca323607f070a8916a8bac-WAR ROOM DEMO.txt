  if (!proposal) return res.status(404).json({ error: "Proposal not found" });

  const currentHash = createClaimHash(current_claim, current_evidence);
  
  let decision = "EXECUTED";
  let reason_code = "SUCCESS";
  let reason_detail = "All invariants satisfied.";
  let judge_output = null;
  let judge_type = "deterministic";
  let should_burn_token = false;

  // INVARIANT 1: Token Authenticity (Non-Fatal Refusal)
  if (proposal.execution_token !== execution_token) {
    return res.status(200).json({ 
      decision: "REFUSED", 
      reason_code: "INVALID_TOKEN", 
      detail: "Execution token mismatch. Proposal state preserved." 
    });
  }

  // INVARIANT 2: Replay Protection (State-Based Refusal)

      reason_code: "REPLAY_ATTACK", 
      detail: `Proposal already finalized at ${proposal.token_used_at}.` 
    });
  }

  // INVARIANT 3: Strict Hash Drift (Fatal Refusal)
  if (proposal.evidence_hash !== currentHash) {
    decision = "REFUSED";
    reason_code = "DRIFT_HASH";
    reason_detail = "Byte-level mismatch detected. Integrity compromised.";
    should_burn_token = true;
  } 
  
  // INVARIANT 4: Semantic Drift (The Compliance Officer)
  else if (process.env.SEMANTIC_DRIFT === 'on') {

    try {
      const model = genAI.getGenerativeModel({ model: "gemini-1.5-flash" });
      
      // Clean, conservative system instruction for compliance
      const prompt = `You are a compliance drift gate. 
      Compare Original: "${proposal.claim}" vs Current: "${current_claim}".
      If polarity, scope (some vs all), or quantities change, drift_detected=true.
      Output JSON only. JSON:`;
      
      const result = await model.generateContent(prompt);
      let responseText = result.response.text().trim();
      
      // Safety: Strip markdown code fences
      responseText = responseText.replace(/^```json|```$/g, "");
      const parsed = JSON.parse(responseText);
      
      judge_output = JSON.stringify(parsed);
      
      if (parsed.drift_detected) {
        decision = "REFUSED";

        reason_detail = `Compliance Refusal (${parsed.drift_type}): ${parsed.explanation}`;
        should_burn_token = true;
      }
    } catch (e) {

      return res.status(500).json({ 
        decision: "REFUSED", 
        reason_code: "JUDGE_UNAVAILABLE", 
        detail: "Semantic judge failed to respond. Proposal state preserved." 
      });

    }
  })();

// Hardened JSON recovery for LLM output
function safeParseJSON(text) {
  // Extract the first {...} block defensively
  const match = String(text || "").match(/\{[\s\S]*\}/);
  if (!match) throw new Error("No JSON object found in judge output");
  return JSON.parse(match[0]);
}

// Strict-ish canonicalization for claim+evidence hash
function createClaimHash(claim, evidence) {
  const ev = Array.isArray(evidence) ? evidence.map(String) : [];
  const normalized = `${String(claim || "").trim().toLowerCase()}|${ev
    .slice()
    .sort()
    .join(",")
    .toLowerCase()}`;
  return crypto.createHash("sha256").update(normalized).digest("hex");
}

// Basic input guards
function validateProposeBody(body) {
  const { claim, evidence } = body || {};
  if (!claim || typeof claim !== "string" || claim.trim().length < 5) {
    return "Invalid claim. Must be a string (min 5 chars).";
  }
  if (!Array.isArray(evidence) || !evidence.every((x) => typeof x === "string")) {
    return "Evidence must be an array of strings.";
  }
  return null;
}

            // Technical failure: preserve token so demo can retry
            reason_code = "JUDGE_UNAVAILABLE";
            reason_detail = "Semantic judge failed. Proposal preserved for retry.";
          }
        }
      }
    }
  } catch (err) {
    reason_code = "SYSTEM_FAULT";
    reason_detail = err.message || String(err);
  }

  // Mandatory audit: all paths create receipts (refusal is success)  [oai_citation:12‡EPISTEMIC LENSE ENGINE.txt](sediment://file_000000002ca4720cb36e782eae430b64)
  db.transaction(() => {
    db.prepare(`
      INSERT INTO receipts (
        id, proposal_id, decision, reason_code, reason_detail,
        input_hash, judge_type, judge_output
      ) VALUES (?, ?, ?, ?, ?, ?, ?, ?)
    `).run(
      receiptId,
      proposal_id || "N/A",
      decision,
      reason_code,
      reason_detail,
      currentHash,
      judge_type,
      judge_output
    );

    if (proposal && should_burn_token) {
      db.prepare(
        "UPDATE proposals SET token_used = 1, token_used_at = CURRENT_TIMESTAMP WHERE id = ?"
      ).run(proposal_id);
    }
  })();

  res.status(200).json({ receipt_id: receiptId, decision, reason_code, detail: reason_detail });
});

// ---- 3) /receipt/:id ----
app.get("/receipt/:id", (req, res) => {
  const receipt = db.prepare("SELECT * FROM receipts WHERE id = ?").get(req.params.id);
  if (!receipt) return res.status(404).json({ error: "Receipt not found" });
  res.json(receipt);
});

## 12) “Things to look out for” (failure modes + fixes)

### 12.1 Hash drift false positives
- strict mode will refuse on any tiny change (that’s the point).
- semantic mode exists for collaborative drafting.

### 12.2 LLM judge unpredictability
- The judge is a *sensor*, not the driver.
- If judge output is invalid JSON or the judge fails → gate remains closed (refusal) and token is preserved (retry).

### 12.3 Token burning rules (demo stability)
- Burn on definitive outcomes (SUCCESS or DRIFT)
- Preserve on technical failure (JUDGE_UNAVAILABLE)
This prevents a malicious or flaky judge from “bricking” the proposal.

### 12.4 Evidence array normalization gotchas
- Evidence is sorted + lowercased in hash. If you supply evidence as different casing/order, strict mode will drift.
- For clean demos: keep evidence identical across propose/execute.

### 12.5 SQLite persistence
- audit.db persists between runs (excellent for showing historical attempts).
- If you want “clean slate,” delete audit.db.
