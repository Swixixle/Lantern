packages/core/src/hashing/stableStringify.ts

/**
 * Deterministic JSON stringify:
 * - sorts object keys
 * - sorts arrays only when caller provides already-sorted arrays
 *   (we do NOT sort arrays generically because order may be semantically meaningful)
 */
export function stableStringify(value: unknown): string {
  return JSON.stringify(sortKeysDeep(value));
}

function sortKeysDeep(x: any): any {
  if (x === null || typeof x !== "object") return x;
  if (Array.isArray(x)) return x.map(sortKeysDeep);
  const keys = Object.keys(x).sort();
  const out: any = {};
  for (const k of keys) out[k] = sortKeysDeep(x[k]);
  return out;
}

packages/core/src/hashing/sha.ts

export async function sha256Hex(input: string): Promise<string> {
  const enc = new TextEncoder();
  const buf = await crypto.subtle.digest("SHA-256", enc.encode(input));
  return [...new Uint8Array(buf)].map(b => b.toString(16).padStart(2, "0")).join("");
}

import { sha256Hex } from "./sha.js";
import { stableStringify } from "./stableStringify.js";
import type { EvidencePack, AnchorSchema, PrimitiveTypeSchema } from "../types/schemas.js";
import { z } from "zod";

type Anchor = z.infer<typeof AnchorSchema>;
type PrimitiveType = z.infer<typeof PrimitiveTypeSchema>;

export async function computeSnippetHash(snippet: string): Promise<string> {
  return sha256Hex(snippet);
}

/**
 * HARD identity: binds to coordinates + snippetHash.
 * If underlying text changes, snippetHash changes => anchorId changes.
 */
export async function computeAnchorId(a: Anchor): Promise<string> {
  const snippetHash = await computeSnippetHash(a.snippet);
  return sha256Hex(stableStringify({
    docId: a.docId,
    segId: a.segId,
    charStart: a.charStart,
    charEnd: a.charEnd,
    snippetHash
  }));
}

/**
 * SOFT identity: binds to anchor + normalized interpretation.
 * Interpretation drift should version-bump, not crash the system.
 */
export async function computeInterpretationId(params: {
  anchorId: string;
  primitiveType: PrimitiveType;
  normalizedValue: string;
}): Promise<string> {
  return sha256Hex(stableStringify(params));
}

/**
 * Utility: locate segment text for a given anchor and re-slice evidence.
 * Used by firewall invariants.
 */
export function resolveAnchorText(pack: EvidencePack, a: Anchor): { segmentText: string; sliced: string } {
  const doc = pack.docs.find(d => d.docId === a.docId);
  if (!doc) throw new Error(`Missing docId ${a.docId}`);
  const seg = doc.segments.find(s => s.segId === a.segId);
  if (!seg) throw new Error(`Missing segId ${a.segId}`);

  const localStart = a.charStart - seg.charStart;
  const localEnd = a.charEnd - seg.charStart;

  if (localStart < 0 || localEnd > seg.text.length || localStart >= localEnd) {
    throw new Error(`Bad offsets for ${a.docId}/${a.segId}: ${a.charStart}-${a.charEnd}`);
  }

  return { segmentText: seg.text, sliced: seg.text.slice(localStart, localEnd) };
}

import type { EvidencePack, ActionProposal, Policy } from "../types/schemas.js";
import { stableStringify } from "./stableStringify.js";

/**
 * Canonicalize pack for hashing.
 * Rule: exclude volatile fields and deep logs.
 * Include only what should define "state fidelity".
 */
export function canonicalizePackForHash(pack: EvidencePack): unknown {
  const docs = [...pack.docs]
    .map(d => ({
      docId: d.docId,
      segments: [...d.segments]
        .map(s => ({ segId: s.segId, charStart: s.charStart, charEnd: s.charEnd, text: s.text }))
        .sort((a,b)=>a.segId.localeCompare(b.segId))
    }))
    .sort((a,b)=>a.docId.localeCompare(b.docId));

  const items = [...pack.items]
    .map(i => ({
      anchorId: i.anchorId,
      interpretationId: i.interpretationId,
      interpretationVersion: i.interpretationVersion,
      primitiveType: i.primitiveType,
      normalizedValue: i.normalizedValue ?? String(i.value),
      anchors: [...i.anchors]
        .map(a => `${a.docId}:${a.segId}:${String(a.charStart).padStart(10,"0")}:${String(a.charEnd).padStart(10,"0")}`)
        .sort(),
      status: i.status,
      flags: i.flags
    }))
    .sort((a,b)=>a.anchorId.localeCompare(b.anchorId));

  return {
    schemaVersion: pack.schemaVersion,
    packId: pack.packId,
    parentRevisionId: pack.parentRevisionId,
    docs,
    items,
    flags: pack.flags
  };
}

/**
 * Canonicalize proposal for hashing.
 * Remove requestedAt to avoid timestamp drift.
 */
export function canonicalizeProposalForHash(p: ActionProposal): unknown {
  const base: any = {
    actionType: p.actionType,
    requestedBy: p.requestedBy,
    correlationId: p.correlationId ?? null,
    payload: p.payload
  };

  if (p.actionType === "PROMOTE_CLAIM") {
    return { ...base, payload: { ...p.payload, anchorIds: [...p.payload.anchorIds].sort() } };
  }
  if (p.actionType === "WRITE_PATCH") {
    // keep file order as provided (semantically meaningful), but ensure stable object keys via stableStringify later
    return base;
  }
  return base;
}

export function canonicalizePolicyForHash(policy: Policy): unknown {
  return policy; // stableStringify sorts keys; policy should be deterministic input
}

export function toStableJson(x: unknown): string {
  return stableStringify(x);
}

export type InvariantResult = {

  ok: boolean;
  severity: "HIGH" | "CRITICAL";

export async function inv_identity_stability(pack: EvidencePack): Promise<InvariantResult> {
  // Recompute anchorIds from anchors (hard evidence). Any mismatch => CRITICAL.
  for (const item of pack.items) {
    // For stability, we compute against the first anchor as representative; you may also enforce all anchors.
    for (const a of item.anchors) {
      const expected = await computeAnchorId(a);
      if (expected !== item.anchorId) {

import type { EvidencePack } from "../types/schemas.js";
import type { InvariantResult } from "./identity.js";
import { resolveAnchorText } from "../hashing/identity.js";

/**
 * Anchor Firewall:
 * Validates that the anchored slice corresponds to the item's value/normalizedValue.
 * For "quote" / "text-like" primitives you can require exact match; for others, containment.
 */
export function inv_anchor_firewall(pack: EvidencePack): InvariantResult {
  for (const item of pack.items) {
    const expectedValue = item.normalizedValue ?? String(item.value);

    for (const a of item.anchors) {
      const { sliced } = resolveAnchorText(pack, a);

      // strict behavior for quote-like primitives
      if (item.primitiveType === "quote") {
        if (sliced !== String(item.value)) {
          return {
            id: "inv.anchor_firewall",
            ok: false,
            severity: "CRITICAL",
            detail: `Quote mismatch for anchor ${item.anchorId}: slice != value`
          };
        }
      } else {
        // containment rule (can tighten later)
        const ok = sliced.includes(String(item.value)) || sliced.includes(expectedValue);
        if (!ok) {
          return {
            id: "inv.anchor_firewall",
            ok: false,
            severity: "CRITICAL",
            detail: `Anchor/value mismatch for anchor ${item.anchorId}: value not supported by slice`
          };
        }
      }
    }
  }

  return { id: "inv.anchor_firewall", ok: true, severity: "CRITICAL" };
}

import type { EvidencePack, ActionProposal, Policy } from "../types/schemas.js";
import { inv_identity_stability, type InvariantResult } from "../invariants/identity.js";
import { inv_anchor_firewall } from "../invariants/anchorFirewall.js";
import { inv_claim_gate_bias } from "../invariants/biasGate.js";

export type GovernorDecision = {
  decision: "APPROVE" | "REFUSE";
  invariantResults: InvariantResult[];
  refusalReasons: string[];
};

export async function evaluateProposal(args: {
  proposal: ActionProposal;
  pack: EvidencePack;
  policy: Policy;
}): Promise<GovernorDecision> {
  const { proposal, pack, policy } = args;

  const results: InvariantResult[] = [];

  results.push(await inv_identity_stability(pack));
  results.push(inv_anchor_firewall(pack));
  results.push(inv_claim_gate_bias(proposal, pack, policy));

  // Policy checks (minimal MVP)
  results.push(inv_opt_in(proposal, pack, policy));
  results.push(inv_write_patch_policy(proposal, policy));
  results.push(inv_execute_downstream_policy(proposal, policy));

  const failed = results.filter(r => !r.ok);
  const refusalReasons = failed.map(r => `${r.id}: ${r.detail ?? "failed"}`);

  return {
    decision: failed.length === 0 ? "APPROVE" : "REFUSE",
    invariantResults: results,
    refusalReasons
  };
}

function inv_opt_in(proposal: ActionProposal, pack: EvidencePack, policy: Policy): InvariantResult {
  if (proposal.actionType !== "PROMOTE_CLAIM") return { id: "inv.opt_in", ok: true, severity: "CRITICAL" };
  if (!policy.requireHumanOptInForConstructive) return { id: "inv.opt_in", ok: true, severity: "CRITICAL" };

  const ok = policy.optIn.constructiveEnabledForPackIds.includes(pack.packId);
  if (!ok) {
    return { id: "inv.opt_in", ok: false, severity: "CRITICAL", detail: `No opt-in for packId ${pack.packId}` };
  }
  return { id: "inv.opt_in", ok: true, severity: "CRITICAL" };
}

function inv_write_patch_policy(proposal: ActionProposal, policy: Policy): InvariantResult {
  if (proposal.actionType !== "WRITE_PATCH") return { id: "inv.write_patch_policy", ok: true, severity: "CRITICAL" };
  if (!policy.actionPermissions.WRITE_PATCH.enabled) {
    return { id: "inv.write_patch_policy", ok: false, severity: "CRITICAL", detail: "WRITE_PATCH disabled by policy" };
  }
  if (!policy.actionPermissions.WRITE_PATCH.allowedRepos.includes(proposal.payload.repo)) {
    return { id: "inv.write_patch_policy", ok: false, severity: "CRITICAL", detail: `Repo not allowed: ${proposal.payload.repo}` };
  }
  return { id: "inv.write_patch_policy", ok: true, severity: "CRITICAL" };
}

function inv_execute_downstream_policy(proposal: ActionProposal, policy: Policy): InvariantResult {
  if (proposal.actionType !== "EXECUTE_DOWNSTREAM") return { id: "inv.execute_downstream_policy", ok: true, severity: "CRITICAL" };
  if (!policy.actionPermissions.EXECUTE_DOWNSTREAM.enabled) {
    return { id: "inv.execute_downstream_policy", ok: false, severity: "CRITICAL", detail: "EXECUTE_DOWNSTREAM disabled by policy" };
  }
  const rank = (x: string) => x === "READ_ONLY" ? 0 : x === "WRITE_LOW" ? 1 : 2;
  if (rank(proposal.payload.effectClass) > rank(policy.actionPermissions.EXECUTE_DOWNSTREAM.maxEffectClass)) {
    return { id: "inv.execute_downstream_policy", ok: false, severity: "CRITICAL", detail: "Effect class exceeds policy" };
  }
  return { id: "inv.execute_downstream_policy", ok: true, severity: "CRITICAL" };
}

import { sha256Hex } from "@loge/core";

export async function signJsonHmac(bodyJson: string, secret: string): Promise<string> {
  const bodyHash = await sha256Hex(bodyJson);
  return sha256Hex(`${secret}|${bodyHash}`);
}

export async function verifyJsonHmac(bodyJson: string, signature: string, secret: string): Promise<boolean> {
  const expected = await signJsonHmac(bodyJson, secret);
  return expected === signature;
}

import Database from "better-sqlite3";

export type DecisionRow = {
  decision_id: string;
  evaluated_at: string;
  action_type: string;
  pack_hash: string;
  payload_hash: string;
  policy_hash: string;
  decision: string;

  token_id: string;
  executed_at: string;
  action_type: string;
  result_json: string;
};

export function openLedger(dbPath = "loge.db") {
  const db = new Database(dbPath);

  db.exec(`
    CREATE TABLE IF NOT EXISTS decisions (
      decision_id TEXT PRIMARY KEY,
      evaluated_at TEXT NOT NULL,
      action_type TEXT NOT NULL,
      pack_hash TEXT NOT NULL,
      payload_hash TEXT NOT NULL,
      policy_hash TEXT NOT NULL,
      decision TEXT NOT NULL,
      token_id TEXT,
      record_json TEXT NOT NULL
    );

    CREATE TABLE IF NOT EXISTS executions (
      token_id TEXT PRIMARY KEY,
      executed_at TEXT NOT NULL,
      action_type TEXT NOT NULL,
      result_json TEXT NOT NULL
    );
  `);

  return {
    putDecision(row: DecisionRow) {
      const stmt = db.prepare(`
        INSERT INTO decisions (decision_id, evaluated_at, action_type, pack_hash, payload_hash, policy_hash, decision, token_id, record_json)
        VALUES (@decision_id, @evaluated_at, @action_type, @pack_hash, @payload_hash, @policy_hash, @decision, @token_id, @record_json)
      `);
      stmt.run(row);
    },
    getDecision(decisionId: string): { record_json: string } | undefined {
      return db.prepare(`SELECT record_json FROM decisions WHERE decision_id = ?`).get(decisionId);
    },
    getExecution(tokenId: string): { result_json: string } | undefined {
      return db.prepare(`SELECT result_json FROM executions WHERE token_id = ?`).get(tokenId);
    },
    putExecution(tokenId: string, actionType: string, result: unknown) {
      const stmt = db.prepare(`
        INSERT INTO executions (token_id, executed_at, action_type, result_json)
        VALUES (?, ?, ?, ?)
      `);
      stmt.run(tokenId, new Date().toISOString(), actionType, JSON.stringify(result));
    }
  };
}

import Fastify from "fastify";
import cors from "@fastify/cors";
import { z } from "zod";

import {
  EvidencePackSchema,
  PolicySchema,
  ActionProposalSchema,
  PermitTokenSchema,
  computePackHash,
  computePayloadHash,
  computePolicyHash,
  evaluateProposal
} from "@loge/core";

import { openLedger } from "./ledger/sqlite.js";
import { randomId } from "./util/hash.js";
import { signJsonHmac, verifyJsonHmac } from "./signing/hmac.js";

const app = Fastify({ logger: true });
await app.register(cors, { origin: true });

const ledger = openLedger(process.env.LOGE_DB_PATH ?? "loge.db");

const SIGNING_SECRET = process.env.LOGE_SIGNING_SECRET ?? "dev_secret_change_me";
const KEY_ID = process.env.LOGE_KEY_ID ?? "dev_key_001";
const TOKEN_TTL_SECONDS = Number(process.env.LOGE_TOKEN_TTL_SECONDS ?? "300");

app.get("/health", async () => ({ ok: true }));

app.post("/propose", async (req, res) => {
  const BodySchema = z.object({
    proposal: ActionProposalSchema,
    pack: EvidencePackSchema,
    policy: PolicySchema
  });

  const parsed = BodySchema.safeParse(req.body);
  if (!parsed.success) return res.status(400).send({ error: "Bad request", detail: parsed.error.flatten() });

  const { proposal, pack, policy } = parsed.data;

  const payloadHash = await computePayloadHash(proposal);
  const packHash = await computePackHash(pack);
  const policyHash = await computePolicyHash(policy);

  const gov = await evaluateProposal({ proposal, pack, policy });

  const decisionId = await randomId(`decision|${packHash}|${payloadHash}|${policyHash}`);
  const now = new Date();
  const expiresAt = new Date(now.getTime() + TOKEN_TTL_SECONDS * 1000).toISOString();

  const tokenBody = {
    tokenId: await randomId(`token|${decisionId}`),
    issuedAt: now.toISOString(),
    expiresAt,
    actionType: proposal.actionType,
    packHash,
    payloadHash,
    policyHash,
    decision: gov.decision,
    requirementsSatisfied: gov.invariantResults.filter(r => r.ok).map(r => r.id),
    refusalReasons: gov.decision === "REFUSE" ? gov.refusalReasons : undefined,
    keyId: KEY_ID
  };

  const signature = await signJsonHmac(JSON.stringify(tokenBody), SIGNING_SECRET);
  const token = { ...tokenBody, signature };

  ledger.putDecision({
    decision_id: decisionId,
    evaluated_at: now.toISOString(),
    action_type: proposal.actionType,
    pack_hash: packHash,
    payload_hash: payloadHash,
    policy_hash: policyHash,
    decision: gov.decision,
    token_id: token.tokenId,
    record_json: JSON.stringify({
      decisionId,
      evaluatedAt: now.toISOString(),
      proposal,
      packHash,
      payloadHash,
      policyHash,
      invariantResults: gov.invariantResults,
      token
    })
  });

  return res.send({
    decisionId,
    decision: gov.decision,
    token,
    invariantResults: gov.invariantResults
  });
});

app.post("/execute", async (req, res) => {
  const BodySchema = z.object({
    proposal: ActionProposalSchema,
    pack: EvidencePackSchema,
    token: PermitTokenSchema
  });

  const parsed = BodySchema.safeParse(req.body);
  if (!parsed.success) return res.status(400).send({ error: "Bad request", detail: parsed.error.flatten() });

  const { proposal, pack, token } = parsed.data;

  // 1) Verify signature
  const { signature, ...body } = token;
  const okSig = await verifyJsonHmac(JSON.stringify(body), signature, SIGNING_SECRET);
  if (!okSig) return res.status(401).send({ error: "Invalid token signature" });

  // 2) Decision & expiry
  if (token.decision !== "APPROVE") return res.status(403).send({ error: "Token is not APPROVE" });
  if (Date.now() > Date.parse(token.expiresAt)) return res.status(403).send({ error: "Token expired" });

  // 3) Drift protection
  const packHashNow = await computePackHash(pack);
  if (packHashNow !== token.packHash) {
    return res.status(409).send({ error: "Evidence drift", expected: token.packHash, actual: packHashNow });
  }

  const payloadHashNow = await computePayloadHash(proposal);
  if (payloadHashNow !== token.payloadHash) {
    return res.status(409).send({ error: "Payload drift", expected: token.payloadHash, actual: payloadHashNow });
  }

  // 4) Replay protection
  const prior = ledger.getExecution(token.tokenId);
  if (prior) return res.status(409).send({ error: "Replay detected", tokenId: token.tokenId });

  // 5) Execute tool adapter (stub)
  const result = await executeToolAdapter(proposal.actionType, (proposal as any).payload);

  ledger.putExecution(token.tokenId, proposal.actionType, result);
  return res.send({ ok: true, result });
});
