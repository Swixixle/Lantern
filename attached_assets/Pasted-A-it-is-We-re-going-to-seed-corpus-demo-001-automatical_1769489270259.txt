// src/storage.ts

export async function seedDemoCorpusIfMissing() {
  const DEMO_CORPUS_ID = "corpus-demo-001";
  const FIXTURE_URL = "/fixtures/test2.pdf"; // ensure this exists in /public/fixtures/

  // 1) Open your IndexedDB (use your existing openDB / getDb helper)
  const db = await getDb(); // <-- replace with your actual db getter

  // 2) Check if corpus exists
  const existing = await db.get("corpora", DEMO_CORPUS_ID); // <-- replace store name if different
  if (existing) return;

  // 3) Create corpus record
  const now = new Date().toISOString();
  await db.put("corpora", {
    corpus_id: DEMO_CORPUS_ID,
    created_at: now,
    updated_at: now,
    title: "Demo Corpus",
  });

  // 4) Fetch fixture PDF bytes
  const resp = await fetch(FIXTURE_URL);
  if (!resp.ok) throw new Error(`Fixture fetch failed: ${FIXTURE_URL} (${resp.status})`);
  const buf = await resp.arrayBuffer();
  const bytes = new Uint8Array(buf);

  // 5) Create a source record (adapt fields to your schema)
  const source_id = crypto.randomUUID();
  const sha256_hex = await sha256Hex(bytes); // use your existing hash helper if you have one

  await db.put("sources", {
    source_id,
    corpus_id: DEMO_CORPUS_ID,
    filename: "test2.pdf",
    mime: "application/pdf",
    bytes,                 // some apps store blob; adapt if you store Blob instead
    sha256_hex,
    created_at: now,
  });

  // 6) Emit ledger/event (optional but recommended if your UI depends on it)
  // If you have an existing "appendEvent" helper, call it here instead.
  await db.add("ledger_events", {
    event_id: crypto.randomUUID(),
    corpus_id: DEMO_CORPUS_ID,
    type: "SOURCE_UPLOADED",
    created_at: now,
    payload: { source_id, filename: "test2.pdf", sha256_hex },
  });

  // 7) Trigger build (anchors/extraction) if you have a build pipeline function.
  // Find your existing "build" / "extract" / "generateAnchors" function and call it here.
  // Example:
  // await runBuildForCorpus(DEMO_CORPUS_ID);

  // If no pipeline exists, we’ll seed minimal anchors next — but try to call your real pipeline first.
}
