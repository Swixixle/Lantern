#### DOM proof script (copy/paste into DevTools console)

Run this on the page you’re validating (Sources page, Claim Space, etc.):

```js
(() => {
  const textExists = (t) =>
    [...document.querySelectorAll("*")].some(el => el.textContent?.trim() === t);

  const linkHrefByText = (t) => {
    const a = [...document.querySelectorAll("a")].find(x => x.textContent?.trim() === t);
    return a ? a.getAttribute("href") : null;
  };

  const tableHeaders = () =>
    [...document.querySelectorAll("table thead th")].map(th => th.textContent.trim());

  const forbiddenPresent = (labels) =>
    labels.map(t => ({
      label: t,
      present: [...document.querySelectorAll("button,a")].some(el => el.textContent?.trim() === t)
    }));

  const result = {
    banner: {
      exists: textExists("Review Mode (Read-Only)")
    },
    nav: {
      claimSpace: linkHrefByText("Claim Space"),
      sources: linkHrefByText("Sources"),
      anchors: linkHrefByText("Anchors"),
      ledger: linkHrefByText("Ledger"),
      snapshots: linkHrefByText("Snapshots")
    },
    sourcesTableHeaders: tableHeaders(),
    forbiddenWriteActionsPresent: forbiddenPresent([
      "Create Corpus",
      "Upload Primary",
      "Upload Secondary",
      "Build Anchors",
      "Save Snapshot",
      "Generate Evidence Packet",
      "Export Corpus Bundle (ZIP)",
      "Export Repro Pack (ZIP)"
    ])
  };

  console.log(JSON.stringify(result, null, 2));
  return result;
})();
```

**Pass conditions for v1.12/v1.11 review-mode UI:**

* `banner.exists === true`
* `nav.*` hrefs are non-null and match the corpusId routes
* `sourcesTableHeaders` includes: `role, filename, sha256_hex, uploaded_at, source_id`
* every `forbiddenWriteActionsPresent[].present === false`
